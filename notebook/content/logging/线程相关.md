### 1) 总览：这个包里有哪些“并发执行单元”

在 `PoseDetectNode` 里，你可以把并发拆成 3 类：

- A. ROS Executor 线程池（由 `MultiThreadedExecutor` 提供）  
    负责跑订阅回调（`Shared_cam0/2_image_callback` → `Sharedimage_callback`）以及参数客户端回调（`param_callback`）。
- B. 两个工作线程（每路相机一个 `std::jthread`）  
    `msg_process_thread1_` / `msg_process_thread2_`，循环 `exec_loopPub` → `mipiImgProcess`，负责“从队列取帧 → 等参数 → `image_process` → 发布”。
- C. DNN 推理/后处理内部线程（`PoseDetection` / `DnnNode` 内部）  
    你在上层看到的是 `Run`/`PostProcess` + `condition_variable` 同步；推理是否另起线程由底层 DNN 框架决定（本包通过 `data_cv_` 等待推理完成）。

```cpp
class PoseDetectNode : public LifecycleTool
{
    // 每一路相机/流（cam0、cam2）都有一份 PoseDetInfo：
    // - 各自一套队列、锁、条件变量、发布器、PoseDetection 实例
    typedef struct PoseDetInfo
    {
        std::unique_ptr<PoseDetection> pose_det;  // 算法对象：内部还有自己的锁/条件变量（DNN 同步）

        // 生产者-消费者队列：生产者是订阅回调，消费者是 mipiImgProcess
        std::condition_variable frame_cv;         // “队列里来了新帧” 的通知
        std::mutex frame_mtx;                     // 保护 img_msg_list 的互斥锁
        std::deque<std::tuple<std::shared_ptr<const char[]>, long, int>> img_msg_list;
        // tuple 里分别是：NV12 数据拷贝指针、frame_time(us)、frame_index

        // 参数相关（注意：这里有 param_cv，但源码里没有 notify）
        std::condition_variable param_cv;         // 用于等待 is_getparam_（但未被显式唤醒）
        
        // 发布器（每路各一份）
        rclcpp::Publisher<custom_interfaces::msg::Keypoint2DArrayList>::SharedPtr kpts_publisher;
        rclcpp::Publisher<std_msgs::msg::Int32MultiArray>::SharedPtr bbox_publisher;
        rclcpp::Publisher<custom_interfaces::msg::HbmMsg1944P>::SharedPtr roi_img_publisher;

        // 统计与打点（线程安全性取决于访问方式；目前多为单线程更新）
        int all_sub_img_num_ = 0;
        int lost_sub_img_num_ = 0;
    } PoseDetInfo;

private:
    std::shared_ptr<PoseDetInfo> pose_det_info1_, pose_det_info2_;

    // 每路一个工作线程（jthread = 可用 stop_token 协作停止）
    std::shared_ptr<std::jthread> msg_process_thread1_ = nullptr;
    std::shared_ptr<std::jthread> msg_process_thread2_ = nullptr;

    std::stop_source stop_source_;  // 析构里 request_stop()

    // 参数与状态：两路共用
    int is_getparam_;               // “参数是否就绪”的全局门控标志
    std::mutex g_mutex_;            // 用来保护 train_* / file_path_ / is_getparam_ 等
};
```

#### 3.1 启动：`handle_activate` 创建 `jthread`

定位

```cpp
int PoseDetectNode::handle_activate()
{
    // 1) 发起异步拉参：回调 param_callback 在 executor 线程里跑
    pc1_socket_client_param_->get_parameters(
        {"chapter", "level", "newFolder_train_data"},
        std::bind(&PoseDetectNode::param_callback, this, std::placeholders::_1));

    // 2) 重置算法对象（注意：这发生在工作线程可能已经在跑的情况下）
    pose_det_info1_->pose_det->reset();
    pose_det_info2_->pose_det->reset();

    // 3) 按 device_mode 起线程。线程函数是 exec_loopPub(pose_det_infoX, stop_token)
    if (device_mode_ == "cam0" || device_mode_ == "dual")
        if (!msg_process_thread1_)
            msg_process_thread1_ = std::make_shared<std::jthread>(
                [this]() { this->exec_loopPub(pose_det_info1_, stop_source_.get_token()); });

    if (device_mode_ == "cam2" || device_mode_ == "dual")
        if (!msg_process_thread2_)
            msg_process_thread2_ = std::make_shared<std::jthread>(
                [this]() { this->exec_loopPub(pose_det_info2_, stop_source_.get_token()); });

    return 0;
}
```

定位

PoseDetectNode::~PoseDetectNode()
{
    stop_source_.request_stop();          // 让 stop_token.stop_requested() 变为 true
    pose_det_info1_->frame_cv.notify_all(); // 唤醒等在 cam0 的 frame_cv 上的线程
    // 注意：这里没有 notify pose_det_info2_->frame_cv（cam2 若在等，可能要靠 10ms 超时醒来）
}

void PoseDetectNode::exec_loopPub(std::shared_ptr<PoseDetInfo>& pose_det_info, std::stop_token stoken)
{
    while (rclcpp::ok() && !stoken.stop_requested())
    {
        mipiImgProcess(pose_det_info, stoken);
    }
}

定位

```cpp
void PoseDetectNode::Sharedimage_callback(std::shared_ptr<PoseDetInfo>& pose_det_info,
                                         const HbmMsg1944P::ConstSharedPtr& img_msg)
{
    // (1) 计算微秒级时间戳
    long frame_time = long(img_msg->time_stamp.sec) * 1000000
                    + (img_msg->time_stamp.nanosec / 1000);

    // (2) 首帧写分辨率到共享 node_info_，并同步到本路 pose_det->node_info_
    //     这里是跨线程共享数据：但只在宽高为 -1 时写入
    if (node_info_->img_width == -1 && node_info_->img_height == -1) {
        node_info_->img_width = img_msg->width;
        node_info_->img_height = img_msg->height;
        pose_det_info->pose_det->node_info_->img_width = node_info_->img_width;
        pose_det_info->pose_det->node_info_->img_height = node_info_->img_height;
    }

    // (3) 参数门控：参数未就绪就不入队，只触发再次拉参
    if (!is_getparam_) {
        pc1_socket_client_param_->get_parameters(
            {"chapter", "level", "newFolder_train_data"},
            std::bind(&PoseDetectNode::param_callback, this, std::placeholders::_1));
        return;
    }

    // (4) 深拷贝 payload：把 ROS 消息里的 data 拷贝到独立 buffer，供异步线程使用
    const char* raw_data = reinterpret_cast<const char*>(img_msg->data.data());
    size_t data_size = img_msg->data.size();
    auto data_copy = std::shared_ptr<const char[]>(new char[data_size], std::default_delete<const char[]>());
    std::copy(raw_data, raw_data + data_size, const_cast<char*>(data_copy.get()));

    // (5) 入队（受 frame_mtx 保护），然后 notify 消费者
    auto item = std::make_tuple(data_copy, frame_time, img_msg->index);
    if (pose_det_info->img_msg_list.size() < MAX_FRAME_NUM) {
        std::unique_lock<std::mutex> lock(pose_det_info->frame_mtx);
        pose_det_info->img_msg_list.push_back(item);
        pose_det_info->frame_cv.notify_all();
    } else {
        // (6) 背压：队列满直接丢帧（生产端不阻塞）
        pose_det_info->lost_sub_img_num_++;
    }

    pose_det_info->all_sub_img_num_++;
}
```

```cpp
void PoseDetectNode::mipiImgProcess(std::shared_ptr<PoseDetInfo>& pose_det_info, std::stop_token stoken)
{
    // A) 从 img_msg_list 取一帧（10ms 轮询式等待）
    std::tuple<std::shared_ptr<const char[]>, long, int> nv12_data_info;
    {
        std::unique_lock<std::mutex> lock(pose_det_info->frame_mtx);
        pose_det_info->frame_cv.wait_for(lock, 10ms, [&] {
            return !rclcpp::ok() || !pose_det_info->img_msg_list.empty() || stoken.stop_requested();
        });

        if (!rclcpp::ok() || pose_det_info->img_msg_list.empty() || stoken.stop_requested())
            return;

        nv12_data_info = pose_det_info->img_msg_list.front();
        pose_det_info->img_msg_list.pop_front(); // 注意：这里先 pop 了
    }

    // B) 参数门控：在 g_mutex_ 下最多等 3 秒
    //    但 pose_det_info->param_cv 没有 notify，因此通常靠“超时醒来”或谓词已满足直接返回
    {
        std::unique_lock<std::mutex> param_lock(g_mutex_);
        pose_det_info->param_cv.wait_for(param_lock, 3s, [this] {
            return is_getparam_ || !rclcpp::ok();
        });

        if (!is_getparam_) {
            // 参数仍没就绪：再次触发拉参，然后 return
            // 由于前面已经 pop，本帧就被丢弃
            pc1_socket_client_param_->get_parameters(...);
            return;
        }
    }

    // C) 算法与发布：这部分通常是本线程最耗时的段
    auto nv12_data = std::get<0>(nv12_data_info);
    long img_time   = std::get<1>(nv12_data_info);
    int frame_index = std::get<2>(nv12_data_info);

    pose_det_info->pose_det->image_process(nv12_data.get(), img_time, frame_index);
    this->publish_msg(pose_det_info, frame_index);
}
```

void PoseDetectNode::param_callback(std::shared_future<std::vector<rclcpp::Parameter>> future_response)
{
    try {
        const auto& response_data = future_response.get();
        if (!response_data.empty()) {
            // 用 g_mutex_ 保护共享数据：file_path_、train_*、is_getparam_
            std::lock_guard<std::mutex> lock(g_mutex_);
            train_chapter_ = response_data[0].as_string();
            train_level_   = response_data[1].as_string();
            file_path_     = response_data[2].as_string();
            is_getparam_   = true;
        }
        // 注意：这里没有 pose_det_info->param_cv.notify_all()
        // → 等待端主要靠 wait_for 超时或谓词本就满足
    } catch (...) { ... }
}

### 7) 这份代码的并发“关键点”总结（最值得你留意的地方）

- 生产/消费队列：`img_msg_list` 用 `frame_mtx` 保护，`frame_cv` 唤醒消费者；队列满丢帧而非阻塞。
- 参数门控是全局的：`is_getparam_` 与 `g_mutex_` 是全局共享；两路相机共用这把门。
- `param_cv` 目前并未发挥“及时唤醒”的作用：没有任何地方 `notify_all` 它，所以等待主要依赖超时。
- 析构只唤醒了 cam0 的 `frame_cv`：cam2 线程若正等待，会靠 10ms 超时循环醒来检查 stop_token。
