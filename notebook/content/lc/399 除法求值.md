from collections import defaultdict

class Solution:
    def bfs(self,node:list, traj, target):
        # 终止条件
        if node not in self.g: return False
        if node == target: return traj

        # 记录已搜索节点，防止环循环
        self.vst.add(node)
        # 深度优先搜索：
        for next,aug in self.g[node]:
            if next in self.vst: continue #已搜索，直接跳过
            result = self.bfs(next, traj*aug, target)
            if result: return result # 找到结果，直接返回
        return False # 当所有子节点都找不到结果时，当前节点返回False

    def calcEquation(self, equations: list, values: list, queries: list) -> list:
        # 先建图，对于equations中的每一对x,y有:
        # g[x] = [(y, r=x/y), ...]
        # g[y] = [(x, r=y/x), ...]
        self.g = defaultdict(list)
        for (x,y),r in zip(equations,values):
            self.g[x].append((y, r))
            self.g[y].append((x, 1/r))

        ans = []
        # 对queries中的每一对x,y，调用dfs进行搜索，并将结果放入ans
        for x,y in queries:
            self.vst = set() #由于图中存在环，因此要排除已visit的node
            temp = self.bfs(node=x,traj=1.,target=y) # traj: 某条trajectory的累积值，默认为1.
            if temp: ans.append(temp)
            else: ans.append(-1.)
        return ans
